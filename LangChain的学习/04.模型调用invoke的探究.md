#### 场景1:Invoke入参是[普通字符串]
```java
import os

from dotenv import load_dotenv
from langchain_openai import ChatOpenAI

"""
   Invoke的用法 - 入参是-字符串
"""

# 读取 .env 配置文件
load_dotenv(override=True)

# 初始化通义千问模型（配置从 .env 文件中读取）
model = ChatOpenAI(
    model=os.getenv("QWEN_MODEL", "qwen-max"),
    openai_api_key=os.getenv("DASHSCOPE_API_KEY"),
    openai_api_base=os.getenv("DASHSCOPE_BASE_URL"),
    temperature=0.7
)

# 调用并输出结果
response = model.invoke("你好嘛")
print(response.content)
```

#### 场景3:Invoke入参是[消息Messsage]
```java
import os

from dotenv import load_dotenv
from langchain_core.messages import SystemMessage, HumanMessage, AIMessage
from langchain_openai import ChatOpenAI

"""
   Invoke的用法 - 入参是-Message
"""

# 读取 .env 配置文件
load_dotenv(override=True)

# 初始化通义千问模型（配置从 .env 文件中读取）
model = ChatOpenAI(
    model=os.getenv("QWEN_MODEL", "qwen-max"),
    openai_api_key=os.getenv("DASHSCOPE_API_KEY"),
    openai_api_base=os.getenv("DASHSCOPE_BASE_URL"),
    temperature=0.7
)

messages = [
    SystemMessage(content="你是一个专业的数学老师"),
    HumanMessage(content="帮我解释一下什么是斐波那契数列"),
]
# 第1次对话
response1 = model.invoke(messages)
print(f"AI的回复1: {response1.content}")

#添加记忆
messages.append(AIMessage(content=response1.content))
messages.append(HumanMessage(content="我刚才问了什么问题？"))

# 第2次对话
response2 = model.invoke(messages)
print(f"AI的回复2: {response2.content}")

```

> AIMessage各个参数分析如下:
```json
AIMessage(
    # ========== 核心字段 ==========
    content='你好！我是一个人工智能助手...',
    # 模型的回复文本内容，这是最重要的字段
    # 日常使用中 90% 的情况只需要 .content

    # ========== 扩展参数 ==========
    additional_kwargs={'refusal': None},
    # 模型提供商返回的额外参数（非标准字段）
    # refusal: OpenAI 的安全拒绝标记，None 表示未被拒绝
    # 不同提供商可能有不同的额外字段放在这里

    # ========== 响应元数据 ==========
    response_metadata={
        # --- Token 用量（计费核心）---
        'token_usage': {
            'completion_tokens': 41,    # 模型生成的 token 数（输出）
            'prompt_tokens': 10,        # 输入的 token 数（你发的内容）
            'total_tokens': 51,         # 总计 token 数 = 10 + 41
            'completion_tokens_details': None,
            'prompt_tokens_details': {
                'audio_tokens': None,       # 音频输入 token（多模态场景）
                'cache_write_tokens': None, # 写入缓存的 token
                'cached_tokens': 0,         # 命中缓存的 token（0=没有）
                'image_tokens': None,       # 图片输入 token（多模态场景）
                'text_tokens': None         # 纯文本 token
            }
        },

        # --- 模型信息 ---
        'model_provider': 'openai',     # 模型提供商（这里走的是 OpenAI 兼容接口）
        'model_name': 'qwen-max',       # 实际调用的模型名称

        # --- 请求标识 ---
        'system_fingerprint': None,     # OpenAI 的系统指纹（用于追踪模型版本）
        'id': 'chatcmpl-7e65c6d2-...',  # 本次请求的唯一 ID，排查问题时用
        'finish_reason': 'stop',        # 停止原因：stop=正常结束, length=达到长度限制
        'logprobs': None                # token 级别的概率信息（需显式请求才有）
    },

    # ========== LangChain 内部标识 ==========
    id='lc_run--01a0e24d-e5b6-...',
    # LangChain 内部运行 ID，用于链路追踪和调试
    # 与 response_metadata 中的 id 不同，这个是 LangChain 框架层面的

    # ========== 工具调用相关 ==========
    tool_calls=[],
    # 模型请求调用的工具列表
    # 当模型需要调用函数/工具时，这里会填充工具名和参数
    # 例如: [{"name": "search", "args": {"query": "..."}}]

    invalid_tool_calls=[],
    # 模型尝试调用但格式不合法的工具调用
    # 用于容错处理，正常情况为空列表

    # ========== 用量元数据（LangChain 统一格式）==========
    usage_metadata={
        'input_tokens': 10,         # 输入 token 数
        'output_tokens': 41,        # 输出 token 数
        'total_tokens': 51,         # 总 token 数
        'input_token_details': {
            'cache_read': 0         # 缓存读取的 token 数（可省钱）
        },
        'output_token_details': {}  # 输出 token 的详细信息
    }
    # 注意：这是 LangChain 统一抽象后的格式
    # 与 response_metadata.token_usage 是同一份数据的不同表达
)
```
